Create `.github/workflows/playwright.yml` following the pattern from RESEARCH.md:

```yaml
name: Playwright E2E Tests

on:
  push:
    branches: [main]

jobs:
  e2e-tests:
    name: E2E Tests (Chromium Mobile)
    runs-on: ubuntu-latest
    timeout-minutes: 30

    steps:
      - uses: actions/checkout@v4

      - uses: pnpm/action-setup@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install

      - name: Install Playwright Chromium
        run: pnpm --filter @jamia/web exec playwright install chromium --with-deps

      - name: Start Supabase
        working-directory: apps/web
        run: |
          npx supabase start
          npx supabase status

      - name: Run Playwright tests
        working-directory: apps/web
        run: pnpm exec playwright test
        env:
          CI: true
          VITE_SUPABASE_URL: http://127.0.0.1:54421
          VITE_SUPABASE_PUBLISHABLE_KEY: eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZS1kZW1vIiwicm9sZSI6ImFub24iLCJleHAiOjE5ODM4MTI5OTZ9.CRXP1A7WOeoJeXxjNni43kdQwgnWNReilDMblYTn_I0
          VITE_API_URL: http://localhost:3000
          VITE_GOOGLE_MAPS_API_KEY: ''

      - name: Upload test report
        uses: actions/upload-artifact@v4
        if: ${{ !cancelled() }}
        with:
          name: playwright-report
          path: apps/web/playwright-report/
          retention-days: 30
```

Key details:
- Triggers on push to main only (per CONTEXT decisions -- free tier friendly)
- Uses pnpm/action-setup@v4 (auto-detects version from packageManager field)
- Installs only Chromium (per CONTEXT decisions -- Chromium only)
- `--with-deps` installs OS-level dependencies needed by Chromium on Ubuntu
- Starts Supabase locally in CI (uses Docker, which is available on ubuntu-latest)
- Sets VITE_GOOGLE_MAPS_API_KEY to empty string to prevent Google Maps calls
- Uploads HTML report as artifact for debugging failures
- 30-minute timeout to prevent runaway CI costs
- The webServer config in playwright.config.ts handles starting the Vite dev server
- The NestJS backend needs to be started too -- the webServer config should handle this via `pnpm dev` which starts both via Turborepo. If the Turborepo `dev` command starts both apps, this works. If not, add a step to start the backend separately.
