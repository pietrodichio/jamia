```typescript
// ... existing searchEvents method ...

// Build filter parameters for RPC function or direct query
let query = this.supabase
  .from('events')
  .select('*, profiles:owner_id(id, name)')
  .eq('status', 'published');

// ... existing filters (location, type, keyword, date_from, date_to) ...

// Add tag filters using GIN index containment operator
if (searchDto.tags && searchDto.tags.length > 0) {
  query = query.contains('tags', searchDto.tags);
}

if (searchDto.accommodation_options && searchDto.accommodation_options.length > 0) {
  query = query.contains('accommodation_options', searchDto.accommodation_options);
}

if (searchDto.food_options && searchDto.food_options.length > 0) {
  query = query.contains('food_options', searchDto.food_options);
}

// Add teacher filter using event_teachers join
if (searchDto.teacher_id) {
  // Use inner join to filter events where teacher_id matches
  const { data: teacherEventIds, error: teacherError } = await this.supabase
    .from('event_teachers')
    .select('event_id')
    .eq('user_id', searchDto.teacher_id);

  if (teacherError) {
    throw new Error(`Failed to query teachers: ${teacherError.message}`);
  }

  const eventIds = teacherEventIds.map(t => t.event_id);
  if (eventIds.length > 0) {
    query = query.in('id', eventIds);
  } else {
    // No events for this teacher, return empty result
    return [];
  }
}

// ... rest of method (order by, execute query, return results) ...
```

**Note on contains operator:** Supabase client's `.contains()` method translates to PostgreSQL's `@>` operator, which uses the GIN index created in migration 05-03.

**Note on teacher filter:** Uses separate query to event_teachers table, then filters events by matching IDs. This is more efficient than LEFT JOIN for single-teacher queries.

Research reference: PostgreSQL ARRAY @> operator with GIN indexes provides 10-50x performance vs junction tables (RESEARCH.md Pattern 6).
