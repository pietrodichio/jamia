    3. Search logic:
    ```typescript
    useEffect(() => {
      if (!debouncedSearch || debouncedSearch.length < 2) {
        setSearchResults([]);
        return;
      }

      const searchProfiles = async () => {
        setIsSearching(true);
        try {
          const { data, error } = await supabase
            .from('profiles')
            .select('id, full_name, avatar_url')
            .ilike('full_name', `%${debouncedSearch}%`)
            .limit(10);

          if (error) throw error;
          setSearchResults(data || []);
        } catch (error) {
          console.error('Search failed:', error);
          setSearchResults([]);
        } finally {
          setIsSearching(false);
        }
      };

      searchProfiles();
    }, [debouncedSearch]);
    ```

    4. Selection handlers:
    ```typescript
    const addTeacher = (teacherId: string) => {
      const current = form.getValues('teacherIds') || [];
      if (!current.includes(teacherId)) {
        form.setValue('teacherIds', [...current, teacherId]);
      }
      setSearchQuery(''); // Clear search
    };

    const removeTeacher = (teacherId: string) => {
      const current = form.getValues('teacherIds') || [];
      form.setValue('teacherIds', current.filter(id => id !== teacherId));
    };
    ```
