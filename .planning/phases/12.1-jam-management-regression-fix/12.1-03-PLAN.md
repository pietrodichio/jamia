With:

```typescript
async getJamById(idParam: string, userId?: string, isSuperAdmin = false) {
  // First, try direct lookup by jam ID
  let { data, error } = await this.supabase
    .from('jams')
    .select(`
      *,
      jam_participants (
        id,
        state
      )
    `)
    .eq('id', idParam)
    .maybeSingle();

  // If not found by jam ID, try lookup by source_event_id
  // This enables /jam/:eventId routes to work for event-based jams
  if (!data && (!error || error.code === 'PGRST116')) {
    const result = await this.supabase
      .from('jams')
      .select(`
        *,
        jam_participants (
          id,
          state
        )
      `)
      .eq('source_event_id', idParam)
      .maybeSingle();

    data = result.data;
    error = result.error;
  }

  if (!data) {
    throw new NotFoundException('Jam not found');
  }

  // ... rest of existing logic (permission checks, participant counts, etc.)
}
```

Key changes:
- Use maybeSingle() instead of single() for first query (avoids error on no match)
- Add fallback query using source_event_id
- Only throw NotFoundException if both lookups fail
